# === Model definition at MODULE scope ===
# If you are using a custom nn.Module, define the CLASS here:
class CNNGRUHybrid(nn.Module):
    def __init__(self, num_classes, dropout=0.3):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 32, 3, padding=1)
        self.bn1 = nn.BatchNorm2d(32)
        self.conv2 = nn.Conv2d(32, 64, 3, padding=1)
        self.bn2 = nn.BatchNorm2d(64)
        self.conv3 = nn.Conv2d(64, 128, 3, padding=1)
        self.bn3 = nn.BatchNorm2d(128)
        self.pool = nn.MaxPool2d(2)
        self.relu = nn.ReLU(inplace=True)
        
        # GRU layer
        self.gru = nn.GRU(128, 128, num_layers=2, batch_first=True, dropout=dropout if 2 > 1 else 0)
        
        # Head
        self.dropout = nn.Dropout(dropout)
        self.head = nn.LazyLinear(num_classes)
        
    def forward(self, x):
        # CNN front-end
        x = self.pool(self.relu(self.bn1(self.conv1(x))))
        x = self.pool(self.relu(self.bn2(self.conv2(x))))
        x = self.pool(self.relu(self.bn3(self.conv3(x))))
        
        # Reshape for GRU
        batch_size, channels, height, width = x.size()
        x = x.transpose(1, 2).contiguous()  # (batch, height, channels, width)
        x = x.view(batch_size, height, -1)  # (batch, height, channels*width)
        
        # GRU
        x, _ = self.gru(x)
        
        # Take last time step
        x = x[:, -1, :]  # (batch, features)
        
        # Head
        x = self.dropout(x)
        x = self.head(x)
        
        return x
